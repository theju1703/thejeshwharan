🧠 CNN Architecture

model = Sequential([ Conv2D(32, (3,3), activation='relu', input_shape=(150,150,3)), MaxPooling2D(2,2), Conv2D(64, (3,3), activation='relu'), MaxPooling2D(2,2), Conv2D(128, (3,3), activation='relu'), MaxPooling2D(2,2), Conv2D(128, (3,3), activation='relu'), MaxPooling2D(2,2), Flatten(), Dropout(0.5), Dense(512, activation='relu'), Dense(6, activation='softmax') ])
